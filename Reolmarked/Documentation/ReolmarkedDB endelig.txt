-- Opret database
CREATE DATABASE ReolmarkedDB;
GO

USE ReolmarkedDB;
GO


-- Opret PaymentMethod
CREATE TABLE PaymentMethod (
    PaymentMethodId INT IDENTITY(1,1) PRIMARY KEY,
    PaymentMethodName NVARCHAR(30) NOT NULL
);
GO

-- Opret RackStatus
CREATE TABLE RackStatus (
    RackStatusId INT IDENTITY(1,1) PRIMARY KEY,
    StatusName NVARCHAR(30) NOT NULL
);
GO

-- Opret Rack
CREATE TABLE Rack (
    RackId INT PRIMARY KEY,
    RackStatusId INT NOT NULL,
    FOREIGN KEY (RackStatusId) REFERENCES RackStatus(RackStatusId)
);
GO

-- Opret Renter
CREATE TABLE Renter (
    RenterId INT IDENTITY(1,1) PRIMARY KEY,
    FirstName NVARCHAR(30) NOT NULL,
    LastName NVARCHAR(30) NOT NULL,
    Phone NVARCHAR(20) UNIQUE,
    Email NVARCHAR(50) UNIQUE,
    PaymentMethodId INT NOT NULL,
    FOREIGN KEY (PaymentMethodId) REFERENCES PaymentMethod(PaymentMethodId)
);
GO

-- Opret RentalContract
CREATE TABLE RentalContract (
    RentalContractId INT IDENTITY(1,1) PRIMARY KEY,
    StartDate DATE NOT NULL,
    EndDate DATE NULL,
    RenterId INT NOT NULL,
    RackId INT NOT NULL,
    FOREIGN KEY (RenterId) REFERENCES Renter(RenterId),
    FOREIGN KEY (RackId) REFERENCES Rack(RackId),
    CHECK (EndDate > StartDate)
);
GO

-- Opret SaleLine
CREATE TABLE SaleLine (
    SaleLineId INT IDENTITY(1,1) PRIMARY KEY,
    SaleDate DATE NOT NULL,
    Price DECIMAL(8,2) NOT NULL CHECK (Price >= 0),
    RackId INT NOT NULL,
    FOREIGN KEY (RackId) REFERENCES Rack(RackId)
);
GO

-- Opret 80 reoler (hvis de ikke allerede findes)
DECLARE @rackId INT = 1;
WHILE @rackId <= 80
BEGIN
    IF NOT EXISTS (SELECT 1 FROM Rack WHERE RackId = @rackId)
    BEGIN
        INSERT INTO Rack (RackId, RackStatusId) VALUES (@rackId, 1);
    END
    SET @rackId += 1;
END;
GO

-- Randomisér 43 reoler som optaget
;WITH Randomized AS (
    SELECT RackId, ROW_NUMBER() OVER (ORDER BY NEWID()) AS rn
    FROM Rack
)
UPDATE R
SET RackStatusId = CASE WHEN rn <= 43 THEN 2 ELSE 1 END
FROM Rack R
JOIN Randomized RR ON R.RackId = RR.RackId;
GO

-- Indsæt 43 kontrakter på de optagede reoler
DECLARE @Occupied TABLE (RowNum INT, RackId INT);

INSERT INTO @Occupied
SELECT ROW_NUMBER() OVER (ORDER BY NEWID()), RackId
FROM Rack WHERE RackStatusId = 2;

DECLARE @i INT = 1;
WHILE @i <= 43
BEGIN
    DECLARE @rack INT = (SELECT RackId FROM @Occupied WHERE RowNum = @i);
    DECLARE @renter INT = ((@i - 1) % (SELECT COUNT(*) FROM Renter)) + 1;
    DECLARE @startDate DATE = DATEADD(MONTH, ((@i - 1) % 5), '2025-05-01');

    INSERT INTO RentalContract (StartDate, EndDate, RenterId, RackId)
    VALUES (@startDate, NULL, @renter, @rack);

    SET @i += 1;
END;
GO

-- Indsæt 50 salgsdata for september fordelt på optagede reoler
SET NOCOUNT ON;
DECLARE @j INT = 1;
WHILE @j <= 50
BEGIN
    DECLARE @rack INT = (
        SELECT RackId FROM (
            SELECT RackId, ROW_NUMBER() OVER (ORDER BY NEWID()) AS rn
            FROM Rack WHERE RackStatusId = 2
        ) t WHERE rn = ((@j - 1) % 43) + 1
    );

    DECLARE @day INT = ((@j - 1) % 30) + 1;
    DECLARE @price DECIMAL(8,2) = ROUND(50 + RAND() * 150, 2);

    INSERT INTO SaleLine (SaleDate, Price, RackId)
    VALUES (DATEFROMPARTS(2025, 9, @day), @price, @rack);

    SET @j += 1;
END;
